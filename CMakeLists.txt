cmake_minimum_required(VERSION 3.16)
project(yolo26_ros LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(Threads REQUIRED)
find_package(OpenCV 4.11 REQUIRED COMPONENTS core imgproc imgcodecs dnn videoio)
message(STATUS "OpenCV ${OpenCV_VERSION} from ${OpenCV_DIR}")
add_library(yolo26_core STATIC src/yolo26_detector.cpp src/file_watcher.cpp)
target_include_directories(yolo26_core PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include> ${OpenCV_INCLUDE_DIRS})
target_link_libraries(yolo26_core PUBLIC opencv_core opencv_imgproc opencv_imgcodecs opencv_dnn Threads::Threads)
add_executable(yolo26_cli src/yolo26_cli.cpp)
set_target_properties(yolo26_cli PROPERTIES OUTPUT_NAME yolo26_detect)
target_link_libraries(yolo26_cli PRIVATE yolo26_core)
add_executable(detector_node src/detector_node.cpp)
target_link_libraries(detector_node yolo26_core)
ament_target_dependencies(detector_node rclcpp)
add_executable(video_feeder_node src/video_feeder_node.cpp)
target_link_libraries(video_feeder_node yolo26_core opencv_videoio)
ament_target_dependencies(video_feeder_node rclcpp)
install(TARGETS yolo26_core ARCHIVE DESTINATION lib)
install(TARGETS yolo26_cli detector_node video_feeder_node DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY include/ DESTINATION include)
install(DIRECTORY launch DESTINATION share/${PROJECT_NAME})
ament_export_include_directories(include)
ament_export_libraries(yolo26_core)
ament_package()
